 # ===== SETTINGS ===== 
$START_DATE = Get-Date "2026-03-20"  
$END_DATE   = Get-Date       # today   
$COMMITS_PER_DAY = 3         #           
$FILE = "index.html"         # file to modify      
      
# ===== LOOP =====    
$current = $START_DATE   
     
while ($current -le $END_DATE) {    

    for ($i=1; $i -le $COMMITS_PER_DAY; $i++) { 
  
        # Add something unique so Git  
        Add-Content $FILE ("<!-- Commit $i on " + $current.ToString("yyyy-MM-dd") + " -->")

        # ISO 8601 format for Git
        $dateString = $current.ToString("yyyy-MM-dd") + "T12:$i:00"

        $env:GIT_AUTHOR_DATE = $dateString
        $env:GIT_COMMITTER_DATE = $dateString

        git add $FILE
        git commit -m "Commit $i on $($current.ToString('yyyy-MM-dd'))"
    }

    $current = $current.AddDays(1)
}
